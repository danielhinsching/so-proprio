blueman               # ícone e gerenciador de Bluetooth

fonts-inter           # fonte limpa, a mais próxima da do macOS/Windows 11
brightnessctl         # teclas de brilho do teclado
cbatticon             # ícone de bateria com os ícones do tema (o do LXQt é um gráfico de pizza)
screengrab            # captura de tela (tecla Print)
light-locker          # tela de bloqueio (Super+L), usa o LightDM
